set(reference_path "${CMAKE_CURRENT_SOURCE_DIR}/../reference/resid-digital.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${reference_path}")
file(READ "${reference_path}" reference)
string(JSON count LENGTH "${reference}")
math(EXPR last "${count} - 1")
set(vectors "")
foreach(index RANGE ${last})
  string(APPEND vectors "{")
  foreach(field sampleRate frequency samples phase lfsr)
    string(JSON value GET "${reference}" ${index} ${field})
    string(APPEND vectors "${value}u,")
  endforeach()
  string(APPEND vectors "},\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference.h" "${vectors}")
add_executable(sidgpu-test engine.c)
target_include_directories(sidgpu-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sidgpu-test PRIVATE sidgpu)
add_test(NAME host-engine COMMAND sidgpu-test)
